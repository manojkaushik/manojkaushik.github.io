Stack Operations Using the Minimum Number of Queues (1 Queue)
========================================
 C Program
========================================

#include <stdio.h>
#define MAX 5

int queue[MAX];
int front = 0, rear = -1, count = 0;

int isEmpty() {
    if (count == 0)
        return 1;
    else
        return 0;
}

void enqueue(int k) {
    if (count == MAX) {
        printf("Overflow. Cannot enqueue %d.\n", k);
        return;
    }
    rear = (rear + 1) % MAX;
    queue[rear] = k;
    count = count + 1;
}

int dequeue() {
    int val = queue[front];
    front = (front + 1) % MAX;
    count = count - 1;
    return val;
}

void push(int k) {
    int i, n;
    enqueue(k);
    n = count;                    // total elements after adding k

    // rotate queue so that k moves to the front
    for (i = 1; i < n; i++) {
        int val = dequeue();
        enqueue(val);
    }
    printf("%d pushed to stack.\n", k);
}

void pop() {
    if (isEmpty() == 1) {
        printf("Stack Underflow.\n");
        return;
    }
    int val = dequeue();
    printf("%d popped from stack.\n", val);
}

void printStack() {
    int i, idx;
    if (isEmpty() == 1) {
        printf("Stack is empty.\n");
        return;
    }
    printf("Stack (top to bottom): ");
    idx = front;
    for (i = 0; i < count; i++) {
        printf("%d ", queue[idx]);
        idx = (idx + 1) % MAX;
    }
    printf("\n");
}

int main() {
    push(10);
    push(20);
    push(30);
    printStack();

    pop();
    printStack();

    return 0;
}
