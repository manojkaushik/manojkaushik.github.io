#include <stdio.h>
#include <stdlib.h>
#define MAX 100

struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *root = NULL;

struct Node *newNode(int k) {
    struct Node *n = (struct Node *)malloc(sizeof(struct Node));
    n->data = k;
    n->left = NULL;
    n->right = NULL;
    return n;
}



// builds the tree for node i of the array (children are at 2i+1 and 2i+2)
// -1 means NULL, so no node is created for it
struct Node *buildFromArray(int arr[], int n, int i) {
    if (i >= n || arr[i] == -1) {
        return NULL;
    }
    struct Node *node = newNode(arr[i]);
    node->left = buildFromArray(arr, n, 2 * i + 1);
    node->right = buildFromArray(arr, n, 2 * i + 2);
    return node;
}



void createTree(int arr[], int n) {
    root = buildFromArray(arr, n, 0);      // arr[0] becomes the root node
    printf("Tree created. Root = %d\n", root->data);
}



// inserts k at the first vacant place found in level order
void insertNode(int k) {
    struct Node *queue[MAX];
    int front = 0, rear = 0;
    struct Node *n = newNode(k);

    if (root == NULL) {
        root = n;
        printf("%d inserted as root.\n", k);
        return;
    }

    queue[rear] = root;
    rear++;
    while (front < rear) {
        struct Node *temp = queue[front];
        front++;

        if (temp->left == NULL) {
            temp->left = n;
            printf("%d inserted as left child of %d.\n", k, temp->data);
            return;
        }
        queue[rear] = temp->left;
        rear++;

        if (temp->right == NULL) {
            temp->right = n;
            printf("%d inserted as right child of %d.\n", k, temp->data);
            return;
        }
        queue[rear] = temp->right;
        rear++;
    }
}



// deletes the node holding key:
// its value is replaced by the last node (in level order), then that last node is removed
void deleteNode(int key) {
    struct Node *queue[MAX];
    int front = 0, rear = 0;
    struct Node *keyNode = NULL;
    struct Node *last = NULL;
    struct Node *temp;

    if (root == NULL) {
        printf("Tree is empty. Cannot delete.\n");
        return;
    }

    // Step 1: level order traversal to find the key node and the last node
    queue[rear] = root;
    rear++;
    while (front < rear) {
        temp = queue[front];
        front++;

        if (temp->data == key && keyNode == NULL) {
            keyNode = temp;
        }
        last = temp;

        if (temp->left != NULL) {
            queue[rear] = temp->left;
            rear++;
        }
        if (temp->right != NULL) {
            queue[rear] = temp->right;
            rear++;
        }
    }

    if (keyNode == NULL) {
        printf("%d not found. Cannot delete.\n", key);
        return;
    }

    // Step 2: copy the last node's value into the key node
    keyNode->data = last->data;

    // Step 3: remove the last node
    if (last == root) {
        free(root);
        root = NULL;
        printf("%d deleted. Tree is now empty.\n", key);
        return;
    }

    front = 0;
    rear = 0;
    queue[rear] = root;
    rear++;
    while (front < rear) {
        temp = queue[front];
        front++;

        if (temp->left == last) {
            temp->left = NULL;
            break;
        }
        if (temp->right == last) {
            temp->right = NULL;
            break;
        }
        if (temp->left != NULL) {
            queue[rear] = temp->left;
            rear++;
        }
        if (temp->right != NULL) {
            queue[rear] = temp->right;
            rear++;
        }
    }
    free(last);
    printf("%d deleted.\n", key);
}




// height = number of nodes on the longest path from root to a leaf
int findHeight(struct Node *node) {
    if (node == NULL) {
        return 0;
    }
    int lh = findHeight(node->left);
    int rh = findHeight(node->right);
    if (lh > rh)
        return lh + 1;
    else
        return rh + 1;
}




int findSize(struct Node *node) {
    if (node == NULL) {
        return 0;
    }
    return 1 + findSize(node->left) + findSize(node->right);
}




int search(struct Node *node, int key) {
    if (node == NULL) {
        return 0;
    }
    if (node->data == key) {
        return 1;
    }
    if (search(node->left, key) == 1) {
        return 1;
    }
    return search(node->right, key);
}




// prints every node with its children (level order), - means NULL
void printTree() {
    struct Node *queue[MAX];
    int front = 0, rear = 0;

    if (root == NULL) {
        printf("Tree is empty.\n");
        return;
    }

    queue[rear] = root;
    rear++;
    while (front < rear) {
        struct Node *temp = queue[front];
        front++;

        printf("  %d : left = ", temp->data);
        if (temp->left != NULL) {
            printf("%d", temp->left->data);
            queue[rear] = temp->left;
            rear++;
        } else {
            printf("-");
        }

        printf(", right = ");
        if (temp->right != NULL) {
            printf("%d", temp->right->data);
            queue[rear] = temp->right;
            rear++;
        } else {
            printf("-");
        }
        printf("\n");
    }
}



int main() {
    int arr[] = {1, 2, 3, 4, -1, 5, 6, 7, 8, -1, -1, 9, -1, -1, -1};
    int n = 15;

    createTree(arr, n);
    printTree();

    printf("Height = %d\n", findHeight(root));
    printf("Size   = %d\n", findSize(root));

    if (search(root, 7) == 1)
        printf("7 is present in the tree.\n");
    else
        printf("7 is not present in the tree.\n");

    if (search(root, 99) == 1)
        printf("99 is present in the tree.\n");
    else
        printf("99 is not present in the tree.\n");

    insertNode(10);
    printTree();

    deleteNode(2);
    printTree();

    printf("Height = %d\n", findHeight(root));
    printf("Size   = %d\n", findSize(root));

    return 0;
}
