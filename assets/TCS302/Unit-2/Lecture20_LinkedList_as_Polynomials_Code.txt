============================================================
Polynomial representation and addition
============================================================

#include <stdio.h>
#include <stdlib.h>

struct Term {
    int coeff;
    int exp;
    struct Term *next;
};



/* Append a term to the end of a polynomial list. Terms are
   conventionally kept in DECREASING exponent order. */

struct Term *insertTerm(struct Term *head, int coeff, int exp)
{
    struct Term *node = malloc(sizeof(struct Term));
    if (node == NULL) { printf("Allocation failed\n"); exit(1); }
    node->coeff = coeff;
    node->exp = exp;
    node->next = NULL;

    if (head == NULL) return node;

    struct Term *p = head;
    while (p->next != NULL) p = p->next;
    p->next = node;
    return head;
}



/* Print a polynomial in a readable form, handling negative
   coefficients cleanly (no "+ -2x^1" ugliness). */

void displayPoly(struct Term *head)
{
    if (head == NULL) { printf("0\n"); return; }
    int first = 1;
    for (struct Term *p = head; p != NULL; p = p->next)
    {
        if (!first)
            printf(p->coeff < 0 ? " - " : " + ");
        else if (p->coeff < 0)
            printf("-");
        int c = p->coeff < 0 ? -p->coeff : p->coeff;
        printf("%dx^%d", c, p->exp);
        first = 0;
    }
    printf("\n");
}




/* Add two polynomials (both in decreasing exponent order) by
   merging term by term - the same shape as a merge-sort merge. */

struct Term *addPoly(struct Term *p1, struct Term *p2)
{
    struct Term *result = NULL;

    while (p1 != NULL && p2 != NULL)
    {
        if (p1->exp == p2->exp)
        {
            int sum = p1->coeff + p2->coeff;
            if (sum != 0)                       /* cancelling terms store nothing */
                result = insertTerm(result, sum, p1->exp);
            p1 = p1->next;
            p2 = p2->next;
        }
        else if (p1->exp > p2->exp)
        {
            result = insertTerm(result, p1->coeff, p1->exp);
            p1 = p1->next;
        }
        else
        {
            result = insertTerm(result, p2->coeff, p2->exp);
            p2 = p2->next;
        }
    }

    while (p1 != NULL) { result = insertTerm(result, p1->coeff, p1->exp); p1 = p1->next; }
    while (p2 != NULL) { result = insertTerm(result, p2->coeff, p2->exp); p2 = p2->next; }

    return result;
}



void freePoly(struct Term *head)
{
    struct Term *p = head;
    while (p != NULL) { struct Term *next = p->next; free(p); p = next; }
}



int main(void)
{
    /* A(x) = 5x^3 + 2x^1 + 7x^0 */
    struct Term *a = NULL;
    a = insertTerm(a, 5, 3);
    a = insertTerm(a, 2, 1);
    a = insertTerm(a, 7, 0);

    /* B(x) = 3x^3 + 4x^2 - 2x^1 */
    struct Term *b = NULL;
    b = insertTerm(b, 3, 3);
    b = insertTerm(b, 4, 2);
    b = insertTerm(b, -2, 1);

    printf("A(x) = "); displayPoly(a);
    printf("B(x) = "); displayPoly(b);

    struct Term *sum = addPoly(a, b);
    printf("A(x) + B(x) = "); displayPoly(sum);

    freePoly(a);
    freePoly(b);
    freePoly(sum);
    return 0;
}
