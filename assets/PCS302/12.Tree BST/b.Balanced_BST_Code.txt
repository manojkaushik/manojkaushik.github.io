Convert a BST into a Balanced BST - Minimum and Maximum Elements

========================================
 C Program
========================================

#include <stdio.h>
#include <stdlib.h>
#define MAX 100

struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *root = NULL;
int arr[MAX];                       // sorted keys of the tree
int count = 0;

struct Node *newNode(int k) {
    struct Node *n = (struct Node *)malloc(sizeof(struct Node));
    n->data = k;
    n->left = NULL;
    n->right = NULL;
    return n;
}

// normal BST insertion (same as the previous question)
struct Node *insertNode(struct Node *node, int k) {
    if (node == NULL) {
        return newNode(k);
    }
    if (k < node->data) {
        node->left = insertNode(node->left, k);
    } else if (k > node->data) {
        node->right = insertNode(node->right, k);
    }
    return node;
}

// number of nodes on the longest path from root to a leaf
int findHeight(struct Node *node) {
    if (node == NULL) {
        return 0;
    }
    int lh = findHeight(node->left);
    int rh = findHeight(node->right);
    if (lh > rh)
        return lh + 1;
    else
        return rh + 1;
}

void printInorder(struct Node *node) {
    if (node == NULL) {
        return;
    }
    printInorder(node->left);
    printf("%d ", node->data);
    printInorder(node->right);
}

void printPreorder(struct Node *node) {
    if (node == NULL) {
        return;
    }
    printf("%d ", node->data);
    printPreorder(node->left);
    printPreorder(node->right);
}

// Step 1: inorder traversal of a BST gives the keys in sorted order
void storeInorder(struct Node *node) {
    if (node == NULL) {
        return;
    }
    storeInorder(node->left);
    arr[count] = node->data;
    count++;
    storeInorder(node->right);
}

// Step 2: the middle key becomes the root, the left half becomes the
// left subtree and the right half becomes the right subtree
struct Node *buildBalanced(int low, int high) {
    if (low > high) {
        return NULL;
    }
    int mid = (low + high) / 2;
    struct Node *node = newNode(arr[mid]);
    node->left = buildBalanced(low, mid - 1);
    node->right = buildBalanced(mid + 1, high);
    return node;
}

void freeTree(struct Node *node) {
    if (node == NULL) {
        return;
    }
    freeTree(node->left);
    freeTree(node->right);
    free(node);
}

void balanceBST() {
    count = 0;
    storeInorder(root);                     // sorted keys are now in arr[]
    freeTree(root);                         // old tree is no longer needed
    root = buildBalanced(0, count - 1);
}

// smallest key = left-most node
int findMin() {
    struct Node *temp = root;
    while (temp->left != NULL) {
        temp = temp->left;
    }
    return temp->data;
}

// largest key = right-most node
int findMax() {
    struct Node *temp = root;
    while (temp->right != NULL) {
        temp = temp->right;
    }
    return temp->data;
}

int main() {
    int keys[] = {20, 30, 40, 50, 60, 65, 70, 80};
    int n = 8;
    int i;

    // keys inserted in sorted order give a skewed (chain-like) BST
    for (i = 0; i < n; i++) {
        root = insertNode(root, keys[i]);
    }

    printf("Before balancing:\n");
    printf("  Inorder  : ");
    printInorder(root);
    printf("\n  Preorder : ");
    printPreorder(root);
    printf("\n  Height   : %d\n", findHeight(root));

    balanceBST();

    printf("After balancing:\n");
    printf("  Inorder  : ");
    printInorder(root);
    printf("\n  Preorder : ");
    printPreorder(root);
    printf("\n  Height   : %d\n", findHeight(root));

    printf("Minimum element = %d\n", findMin());
    printf("Maximum element = %d\n", findMax());

    return 0;
}
