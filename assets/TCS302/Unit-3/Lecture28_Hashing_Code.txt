============================================================
Lecture 28 : Hash Functions and Hashing
============================================================

Two independent programs, both using the SAME six keys and the
SAME hash function (h(key) = key % 7), so their results are
directly comparable.

  PART A - Separate chaining (array of linked lists)
  PART B - Linear probing (open addressing, plain array)

============================================================
PART A : Separate chaining
============================================================

#include <stdio.h>
#include <stdlib.h>

#define TABLE_SIZE 7

struct Node {
    int key;
    struct Node *next;
};

struct Node *table[TABLE_SIZE];

int hashFunc(int key)
{
    return key % TABLE_SIZE;
}

void insert(int key)
{
    int idx = hashFunc(key);
    struct Node *node = malloc(sizeof(struct Node));
    if (node == NULL) { printf("Allocation failed\n"); exit(1); }
    node->key = key;
    node->next = table[idx];   /* insert at front of the chain */
    table[idx] = node;
    printf("insert(%d) -> index %d\n", key, idx);
}

int search(int key)
{
    int idx = hashFunc(key);
    int probes = 0;
    for (struct Node *p = table[idx]; p != NULL; p = p->next)
    {
        probes++;
        if (p->key == key)
            return probes;
    }
    return -1;
}

void display(void)
{
    for (int i = 0; i < TABLE_SIZE; i++)
    {
        printf("  [%d]: ", i);
        for (struct Node *p = table[i]; p != NULL; p = p->next)
            printf("%d -> ", p->key);
        printf("NULL\n");
    }
}

void freeTable(void)
{
    for (int i = 0; i < TABLE_SIZE; i++)
    {
        struct Node *p = table[i];
        while (p != NULL)
        {
            struct Node *next = p->next;
            free(p);
            p = next;
        }
    }
}

int main(void)
{
    for (int i = 0; i < TABLE_SIZE; i++) table[i] = NULL;

    int keys[] = {23, 12, 17, 9, 24, 15};
    int n = 6;

    for (int i = 0; i < n; i++) insert(keys[i]);

    printf("\nTable after all inserts:\n");
    display();

    printf("\nSearch results:\n");
    printf("  search(9)  -> %d probes\n", search(9));
    printf("  search(24) -> %d probes\n", search(24));
    printf("  search(15) -> %d probes\n", search(15));
    printf("  search(99) -> %d (not found)\n", search(99));

    freeTable();
    return 0;
}

============================================================
PART B : Linear probing (open addressing)
============================================================

#include <stdio.h>

#define TABLE_SIZE 7
#define EMPTY -1

int table[TABLE_SIZE];

int hashFunc(int key)
{
    return key % TABLE_SIZE;
}

int insert(int key)
{
    int idx = hashFunc(key);
    int start = idx;
    int probes = 0;

    while (table[idx] != EMPTY)
    {
        probes++;
        idx = (idx + 1) % TABLE_SIZE;   /* wrap around */
        if (idx == start) return -1;    /* table full */
    }
    table[idx] = key;
    printf("insert(%d) -> home index %d, placed at index %d (%d probe%s)\n",
           key, start, idx, probes, probes == 1 ? "" : "s");
    return idx;
}

int search(int key)
{
    int idx = hashFunc(key);
    int start = idx;
    int probes = 0;

    while (table[idx] != EMPTY)
    {
        probes++;
        if (table[idx] == key) return probes;
        idx = (idx + 1) % TABLE_SIZE;
        if (idx == start) break;
    }
    return -1;
}

void display(void)
{
    printf("  index: ");
    for (int i = 0; i < TABLE_SIZE; i++) printf("%3d", i);
    printf("\n  value: ");
    for (int i = 0; i < TABLE_SIZE; i++)
    {
        if (table[i] == EMPTY) printf("  .");
        else printf("%3d", table[i]);
    }
    printf("\n");
}

int main(void)
{
    for (int i = 0; i < TABLE_SIZE; i++) table[i] = EMPTY;

    int keys[] = {23, 12, 17, 9, 24, 15};
    int n = 6;

    for (int i = 0; i < n; i++) insert(keys[i]);

    printf("\nTable after all inserts:\n");
    display();

    printf("\nSearch results:\n");
    printf("  search(9)  -> %d probes\n", search(9));
    printf("  search(24) -> %d probes\n", search(24));
    printf("  search(15) -> %d probes\n", search(15));
    printf("  search(99) -> %d (not found)\n", search(99));

    return 0;
}
