Infix to Postfix Conversion - Expression Tree, BODMAS Rule
========================================
 C Program
========================================

#include <stdio.h>
#include <stdlib.h>
#define MAX 100

struct Node {
    char data;
    struct Node *left;
    struct Node *right;
};

char opStack[MAX];
int top = -1;

void push(char c) {
    top = top + 1;
    opStack[top] = c;
}

char pop() {
    char c = opStack[top];
    top = top - 1;
    return c;
}

char peek() {
    return opStack[top];
}

int isEmpty() {
    if (top == -1)
        return 1;
    else
        return 0;
}

int isOperator(char c) {
    if (c == '+' || c == '-' || c == '*' || c == '/' || c == '^')
        return 1;
    else
        return 0;
}

// BODMAS: ^ first, then / and *, then + and -
int precedence(char op) {
    if (op == '^')
        return 3;
    if (op == '*' || op == '/')
        return 2;
    if (op == '+' || op == '-')
        return 1;
    return 0;                          // for '('
}

void infixToPostfix(char infix[], char postfix[]) {
    int i, j = 0;

    for (i = 0; infix[i] != '\0'; i++) {
        char ch = infix[i];

        if (ch == '(') {
            push(ch);
        }
        else if (ch == ')') {
            // pop everything until the matching '('
            while (isEmpty() == 0 && peek() != '(') {
                postfix[j] = pop();
                j++;
            }
            if (isEmpty() == 0) {
                pop();                 // remove '('
            }
        }
        else if (isOperator(ch) == 1) {
            // pop operators that have higher (or equal) priority
            // ^ is evaluated right to left, so equal priority is not popped for ^
            while (isEmpty() == 0 && peek() != '(' &&
                   (precedence(peek()) > precedence(ch) ||
                   (precedence(peek()) == precedence(ch) && ch != '^'))) {
                postfix[j] = pop();
                j++;
            }
            push(ch);
        }
        else {
            postfix[j] = ch;           // operand goes straight to the output
            j++;
        }
    }

    // pop the remaining operators
    while (isEmpty() == 0) {
        char c = pop();
        if (c != '(') {
            postfix[j] = c;
            j++;
        }
    }
    postfix[j] = '\0';
}

struct Node *newNode(char c) {
    struct Node *n = (struct Node *)malloc(sizeof(struct Node));
    n->data = c;
    n->left = NULL;
    n->right = NULL;
    return n;
}

// builds the expression tree from a postfix expression
struct Node *buildTree(char postfix[]) {
    struct Node *nodeStack[MAX];
    int t = -1;
    int i;

    for (i = 0; postfix[i] != '\0'; i++) {
        struct Node *n = newNode(postfix[i]);

        if (isOperator(postfix[i]) == 1) {
            n->right = nodeStack[t];   // second operand
            t = t - 1;
            n->left = nodeStack[t];    // first operand
            t = t - 1;
        }
        t = t + 1;
        nodeStack[t] = n;
    }
    return nodeStack[t];
}

void postorder(struct Node *node) {
    if (node == NULL) {
        return;
    }
    postorder(node->left);
    postorder(node->right);
    printf("%c", node->data);
}

int main() {
    char infix[MAX];
    char postfix[MAX];

    printf("Enter infix expression (no spaces): ");
    scanf("%s", infix);

    infixToPostfix(infix, postfix);
    printf("Infix   : %s\n", infix);
    printf("Postfix : %s\n", postfix);

    // the postorder traversal of the expression tree gives the same postfix
    struct Node *root = buildTree(postfix);
    printf("Postorder of expression tree : ");
    postorder(root);
    printf("\n");

    return 0;
}
