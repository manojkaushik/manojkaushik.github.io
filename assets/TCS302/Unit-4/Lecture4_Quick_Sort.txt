#include <stdio.h>

void swap(int *x, int *y)
{
    int t = *x;
    *x = *y;
    *y = t;
}

/* Partition a[lo..hi] around the last element (the pivot).
   Returns the final index of the pivot. */
int partition(int a[], int lo, int hi)
{
    int pivot = a[hi];
    int i = lo - 1;                    /* end of the small zone */

    for (int j = lo; j < hi; j++)
    {
        if (a[j] < pivot)
        {
            i++;
            swap(&a[i], &a[j]);
        }
    }

    swap(&a[i + 1], &a[hi]);           /* pivot goes home */
    return i + 1;
}

void quickSort(int a[], int lo, int hi)
{
    if (lo < hi)
    {
        int p = partition(a, lo, hi);
        quickSort(a, lo, p - 1);       /* left of the pivot  */
        quickSort(a, p + 1, hi);       /* right of the pivot */
    }
}

void printArray(int a[], int n)
{
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
}

int main(void)
{
    int a[] = {7, 3, 9, 1, 5, 8, 4};
    int n = sizeof(a) / sizeof(a[0]);

    printf("Before: ");
    printArray(a, n);

    quickSort(a, 0, n - 1);

    printf("After:  ");
    printArray(a, n);

    return 0;
}
