============================================================
Insertion Sort
============================================================

#include <stdio.h>

void insertionSort(int a[], int n)
{
    for (int i = 1; i < n; i++)
    {
        int key = a[i];          /* element to insert */
        int j = i - 1;

        while (j >= 0 && a[j] > key)
        {
            a[j + 1] = a[j];     /* shift larger element right */
            j--;
        }
        a[j + 1] = key;          /* drop key into the gap */
    }
}

void printArray(int a[], int n)
{
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
}

int main(void)
{
    int a[] = {7, 3, 5, 2, 9, 1};
    int n = sizeof(a) / sizeof(a[0]);

    printf("Before: ");
    printArray(a, n);

    insertionSort(a, n);

    printf("After:  ");
    printArray(a, n);

    return 0;
}
